cmake_minimum_required(VERSION 3.16)
project(AtlasNavigation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_OSX_DEPLOYMENT_TARGET 13.0)
set(V ${CMAKE_CURRENT_SOURCE_DIR}/../Vendor)
file(GLOB RECAST ${V}/recast/Recast/Source/*.cpp ${V}/recast/Detour/Source/*.cpp)
set(COMMON compression config eq_math eqg_loader eqg_model_loader eqg_v4_loader oriented_bounding_box pfs pfs_crc s3d_loader string_util water_map water_map_v1 water_map_v2 wld_fragment zone_map)
set(SOURCES main.cpp ${V}/eqemu/azone/map.cpp ${V}/eqemu/awater/water_map.cpp ${RECAST})
foreach(NAME ${COMMON})
 list(APPEND SOURCES ${V}/eqemu/common/${NAME}.cpp)
endforeach()
file(GLOB LOG ${V}/eqemu/log/*.cpp)
add_executable(AtlasNavigation ${SOURCES} ${LOG})
target_include_directories(AtlasNavigation PRIVATE ${V}/eqemu/common ${V}/eqemu/log ${V}/glm ${V}/json ${V}/recast/Recast/Include ${V}/recast/Detour/Include)
target_compile_definitions(AtlasNavigation PRIVATE DT_POLYREF64=1)
if(MSVC)
 target_compile_options(AtlasNavigation PRIVATE /FI${CMAKE_CURRENT_SOURCE_DIR}/Support.h /utf-8)
 target_compile_definitions(AtlasNavigation PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX _USE_MATH_DEFINES)
else()
 target_compile_options(AtlasNavigation PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/Support.h -Wno-deprecated-declarations)
endif()
set(Z ${V}/zlib)
add_library(atlas_zlib STATIC ${Z}/adler32.c ${Z}/compress.c ${Z}/crc32.c ${Z}/deflate.c ${Z}/infback.c ${Z}/inffast.c ${Z}/inflate.c ${Z}/inftrees.c ${Z}/trees.c ${Z}/uncompr.c ${Z}/zutil.c)
target_include_directories(atlas_zlib PUBLIC ${Z})
target_link_libraries(AtlasNavigation PRIVATE atlas_zlib)
# A consistent path for the Electron launcher, including multi-config generators.
set_target_properties(AtlasNavigation PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set_target_properties(AtlasNavigation PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} ${CMAKE_BINARY_DIR})
endforeach()

file(COPY ${CMAKE_CURRENT_SOURCE_DIR}/Crossings DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
